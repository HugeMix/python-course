1. Переменные и типы данных

# Основные типы данных
x = 10          # int (целое число)
y = 3.14        # float (дробное число)
name = "Алекс"  # str (строка)
flag = True     # bool (True или False)
nothing = None  # NoneType (отсутствие значения)

# Проверка типа
print(type(x))  # <class 'int'>

# Конвертация типов
int("42")        # → 42
float("3.14")    # → 3.14
str(42)          # → "42"
bool(0)          # → False (0, "", [], {} — False)

2. Строки

s = "Привет, мир!"

len(s)              # Длина строки
s.upper()           # ПРИВЕТ, МИР!
s.lower()           # привет, мир!
s.strip()           # Убрать пробелы по краям
s.split(", ")       # ['Привет', 'мир!']
s.replace("мир", "Python")  # Замена
s.find("мир")       # Индекс подстроки (-1 если нет)
"мир" in s          # True — проверка наличия

# f-строки
name = "Алекс"

print(f"Меня зовут {name}, мне {age} лет")

# Методы проверки
"123".isdigit()     # True
"abc".isalpha()     # True
s.startswith("Пр") # True
s.endswith("!")     # True

3. Списки

lst = [1, 2, 3, "четыре", True]

lst[0]          # 1 (первый элемент)
lst[-1]         # True (последний)
lst[1:3]        # [2, 3] (срез)
lst[::-1]       # Реверс списка

lst.append(5)       # Добавить в конец
lst.insert(0, 0)    # Вставить на позицию
lst.remove(2)       # Удалить по значению
lst.pop()           # Удалить и вернуть последний
lst.pop(0)          # Удалить по индексу
len(lst)            # Длина
lst.sort()          # Сортировка
lst.sort(reverse=True)  # Обратная сортировка
sorted(lst)         # Новый отсортированный список
sum(lst)            # Сумма (числа)
min(lst), max(lst)  # Минимум и максимум
lst.count(1)        # Количество вхождений
1 in lst            # True/False — наличие

# Генератор списка
squares = [x**2 for x in range(10)]
evens = [x for x in range(20) if x % 2 == 0]

4. Словари

d = {"name": "Алекс", "age": 25}

d["name"]           # "Алекс"
d.get("city")       # None (без ошибки)
d.get("city", "?")  # "?" (значение по умолчанию)

d["city"] = "Москва"    # Добавить/изменить
del d["age"]            # Удалить ключ

d.keys()            # dict_keys(['name', 'city'])
d.values()          # dict_values(['Алекс', 'Москва'])
d.items()           # пары (ключ, значение)

"name" in d         # True

# Перебор
for key, value in d.items():
    print(f"{key}: {value}")

# Словарный включение
{k: v*2 for k, v in d.items()}

8. Функции

def greet(name, greeting="Привет"):
    """Строка документации."""
    return f"{greeting}, {name}!"

greet("Алекс")              # Привет, Алекс!
greet("Боб", "Здравствуй")  # Здравствуй, Боб!

# Произвольные аргументы
def func(*args, **kwargs):
    print(args)    # Кортеж позиционных аргументов
    print(kwargs)  # Словарь именованных аргументов

# Lambda
double = lambda x: x * 2
square = lambda x: x ** 2

9. Классы

class Person:
    species = "Homo sapiens"  # Атрибут класса

    def __init__(self, name, age):
        self.name = name       # Атрибуты объекта
        self.age = age

    def greet(self):
        return f"Привет, я {self.name}"

    def __str__(self):
        return f"Person({self.name}, {self.age})"

    def __repr__(self):
        return self.__str__()

class Student(Person):  # Наследование
    def __init__(self, name, age, grade):
        super().__init__(name, age)
        self.grade = grade

p = Person("Алекс", 25)
print(p.greet())
print(str(p))

11. Исключения

    result = 10 / 0
except ZeroDivisionError as e:
    print(f"Ошибка: {e}")

    print(f"Другая ошибка: {e}")
else:
    print("Успех!")    # Если не было ошибки
finally:
    print("Выполнится всегда")

raise ValueError("Своя ошибка")
